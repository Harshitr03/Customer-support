# SpotifyCares AI Support Agent — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a runnable, evaluated AI support agent for `@SpotifyCares` that classifies a customer tweet's intent, drafts a reply grounded in historical resolutions, and decides auto-handle vs. escalate with a stated reason — plus the golden set, eval harness, and report that prove it works.

**Architecture:** A small Python package (`src/support_agent`) of focused modules wired by `pipeline.py`. Data prep reconstructs Twitter threads from the raw CSV, filters to Spotify, subsamples ~6k threads, and splits into a disjoint KB corpus and an eval pool. Retrieval is numpy cosine over Gemini embeddings (no vector DB). Three systems (trivial / simple / LLM) are compared for both classification and reply drafting. Escalation is an auditable rule layer. Evaluation is automated metrics + a Gemini LLM-as-judge whose agreement with a human rater is measured via Cohen's kappa. Every LLM/embedding call is disk-cached by input hash so reruns are near-instant.

**Tech Stack:** Python 3.11+, `google-genai` (Gemini 2.5 Flash + `gemini-embedding-001`), pandas, scikit-learn (TF-IDF + LogisticRegression), numpy, scipy (Cohen's kappa / correlation), pyarrow (parquet cache), python-dotenv, pytest.

**Spec:** `docs/superpowers/specs/2026-09-10-spotify-support-agent-design.md`

## Global Constraints

- Python >= 3.11 (tested on 3.13). Single virtualenv, deps pinned in `pyproject.toml`.
- Model IDs (verbatim): generation/classification/judge = `gemini-2.5-flash`; embeddings = `gemini-embedding-001`.
- API key read from `GEMINI_API_KEY` env var (loaded from `.env` via python-dotenv). Never commit `.env`.
- Fixed random seed `SEED = 42` everywhere sampling/splitting happens.
- Subsample working pool = `POOL_SIZE = 6000` threads; corpus/eval split = 90/10 disjoint by thread root id.
- Golden set target size `GOLDEN_SIZE = 200`, stratified across the final intent taxonomy with a per-intent minimum of 10 (fall back to available count if an intent is rarer).
- Simple classification baseline trains TF-IDF + LogisticRegression on **weak-labeled corpus threads** (keyword rules), NOT on golden rows. Decision: keeps the baseline comparison honest.
- Golden labels are LLM-proposed then **author-spot-checked** on a stratified slice; the spot-check correction rate is logged and disclosed in the report. Decision per user.
- All intermediate artifacts (`data/interim/`, `data/cache/`) are gitignored and regenerated by the pipeline. Only `data/golden/` is committed.
- Single source of truth for intents is `src/support_agent/taxonomy.py`; the classifier prompt and the labeling rubric both import from it.
- Every LLM/embedding call goes through `llm_client.py`'s disk cache keyed by a hash of (model, prompt/content, params). Tests must never hit the network — mock the client.

```python
"""Parse twcs.csv, reconstruct threads, filter to Spotify, sample + split."""
import json
import pandas as pd
from . import config

def _load_raw() -> pd.DataFrame:
    df = pd.read_csv(config.RAW_CSV, dtype={"tweet_id": "Int64"})
    df["inbound"] = df["inbound"].astype(str).str.lower().eq("true")
    df["created_at"] = pd.to_datetime(df["created_at"], errors="coerce",
                                      format="%a %b %d %H:%M:%S %z %Y")
    return df

def reconstruct_threads(df: pd.DataFrame) -> list[dict]:
    by_id = {int(r.tweet_id): r for r in df.itertuples() if pd.notna(r.tweet_id)}
    # find roots: tweets with no in_response_to, or whose parent is absent
    def parent(r):
        p = r.in_response_to_tweet_id
        if pd.isna(p) or p in ("", None):
            return None
        try:
            return int(float(p))
        except (ValueError, TypeError):
            return None
    children: dict[int, list[int]] = {}
    for tid, r in by_id.items():
        p = parent(r)
        if p is not None:
            children.setdefault(p, []).append(tid)
    roots = [tid for tid, r in by_id.items() if parent(r) not in by_id]
    threads = []
    for root in roots:
        # BFS collect the connected component
        seen, stack = set(), [root]
        while stack:
            n = stack.pop()
            if n in seen:
                continue
            seen.add(n)
            stack.extend(children.get(n, []))
            pr = parent(by_id[n]) if n in by_id else None
            if pr in by_id and pr not in seen:
                stack.append(pr)
        turns = [by_id[t] for t in seen if t in by_id]
        turns.sort(key=lambda r: (r.created_at is pd.NaT, r.created_at))
        threads.append({
            "root_id": int(min(seen)),
            "turns": [{"tweet_id": int(t.tweet_id), "author_id": str(t.author_id),
                       "inbound": bool(t.inbound), "text": str(t.text)} for t in turns],
        })
    return threads

def spotify_threads(threads: list[dict]) -> list[dict]:
    out = []
    for th in threads:
        replies = [t["text"] for t in th["turns"]
                   if not t["inbound"] and t["author_id"] == config.BRAND]
        if not replies:
            continue
        opens = [t["text"] for t in th["turns"] if t["inbound"]]
        if not opens:
            continue
        out.append({"root_id": th["root_id"], "customer_open": opens[0],
                    "spotify_replies": replies, "turns": th["turns"]})
    return out

def build_pool() -> None:
    config.ensure_dirs()
    corpus_p = config.INTERIM_DIR / "corpus_pool.parquet"
    eval_p = config.INTERIM_DIR / "eval_pool.parquet"
    if corpus_p.exists() and eval_p.exists():
        return
    df = _load_raw()
    sp = spotify_threads(reconstruct_threads(df))
    rows = [{"root_id": t["root_id"], "customer_open": t["customer_open"],
             "spotify_reply": t["spotify_replies"][0],
             "turns_json": json.dumps(t["turns"])} for t in sp]
    pool = pd.DataFrame(rows)
    if len(pool) > config.POOL_SIZE:
        pool = pool.sample(config.POOL_SIZE, random_state=config.SEED)
    pool = pool.sample(frac=1.0, random_state=config.SEED).reset_index(drop=True)
    cut = int(len(pool) * config.CORPUS_FRAC)
    pool.iloc[:cut].to_parquet(corpus_p, index=False)
    pool.iloc[cut:].to_parquet(eval_p, index=False)

def load_pools() -> tuple[pd.DataFrame, pd.DataFrame]:
    return (pd.read_parquet(config.INTERIM_DIR / "corpus_pool.parquet"),
            pd.read_parquet(config.INTERIM_DIR / "eval_pool.parquet"))
```

```python
"""Intent taxonomy for SpotifyCares — single source of truth."""
from dataclasses import dataclass

OTHER = "other"

@dataclass(frozen=True)
class Intent:
    name: str
    definition: str
    examples: tuple[str, ...]

INTENTS: list[Intent] = [
    Intent("playback_bug",
        "App/player not working: crashes, songs won't play, skipping, buffering, offline errors.",
        ("the app keeps crashing when I hit play",
         "songs stop after 10 seconds on my phone")),
    Intent("account_login",
        "Cannot access account: login failures, password reset, hacked/compromised account, email change.",
        ("I can't log into my account", "someone hacked my Spotify account")),
    Intent("billing_subscription",
        "Charges, payment methods, Premium not active after paying, unexpected/duplicate charges.",
        ("I was charged twice for Premium this month",
         "paid for Premium but still seeing ads")),
    Intent("content_catalog",
        "Missing/removed songs, albums, podcasts, wrong metadata, regional availability of content.",
        ("why did my favorite album disappear",
         "this podcast isn't available in my country")),
    Intent("cancellation_refund",
        "Wants to cancel Premium or request a refund for a charge.",
        ("how do I cancel my subscription",
         "I want a refund for last month")),
    Intent("feature_complaint",
        "Dislikes a feature/UX change, shuffle behavior, recommendations, general feedback/complaint.",
        ("the new shuffle is terrible",
         "bring back the old playlist UI")),
    Intent(OTHER,
        "Anything that does not fit the above, including greetings, praise, or off-topic messages.",
        ("you guys are awesome",)),
]

INTENT_NAMES = [i.name for i in INTENTS]

def describe() -> str:
    lines = []
    for it in INTENTS:
        ex = " | ".join(it.examples)
        lines.append(f"- {it.name}: {it.definition} Examples: {ex}")
    return "\n".join(lines)
```

```python
"""Gemini wrapper with on-disk caching. Network calls isolated in _raw_*."""
import hashlib
import json
import os
import numpy as np
from . import config

_client = None

def _get_client():
    global _client
    if _client is None:
        from google import genai
        key = os.environ.get("GEMINI_API_KEY")
        if not key:
            raise RuntimeError("GEMINI_API_KEY not set (see .env.example)")
        _client = genai.Client(api_key=key)
    return _client

def _cache_path(kind: str, key: str):
    config.CACHE_DIR.mkdir(parents=True, exist_ok=True)
    h = hashlib.sha256(key.encode()).hexdigest()[:32]
    return config.CACHE_DIR / f"{kind}_{h}.json"

def _raw_generate(prompt: str, temperature: float, model: str, json_mode: bool) -> str:
    from google.genai import types
    cfg = types.GenerateContentConfig(
        temperature=temperature,
        response_mime_type="application/json" if json_mode else "text/plain",
    )
    resp = _get_client().models.generate_content(model=model, contents=prompt, config=cfg)
    return resp.text or ""

def _raw_embed(texts: list[str]) -> np.ndarray:
    resp = _get_client().models.embed_content(model=config.EMBED_MODEL, contents=texts)
    return np.array([e.values for e in resp.embeddings], dtype=np.float32)

def generate(prompt: str, *, json_mode: bool = False, temperature: float = 0.2,
             model: str | None = None) -> str:
    model = model or config.GEN_MODEL
    key = json.dumps({"m": model, "p": prompt, "t": temperature, "j": json_mode})
    path = _cache_path("gen", key)
    if path.exists():
        return json.loads(path.read_text())["text"]
    text = _raw_generate(prompt, temperature, model, json_mode)
    path.write_text(json.dumps({"text": text}))
    return text

def embed(texts: list[str]) -> np.ndarray:
    out: list[np.ndarray | None] = [None] * len(texts)
    missing_idx, missing_txt = [], []
    for i, t in enumerate(texts):
        p = _cache_path("emb", config.EMBED_MODEL + "::" + t)
        if p.exists():
            out[i] = np.array(json.loads(p.read_text()), dtype=np.float32)
        else:
            missing_idx.append(i); missing_txt.append(t)
    if missing_txt:
        vecs = _raw_embed(missing_txt)
        for j, i in enumerate(missing_idx):
            out[i] = vecs[j]
            _cache_path("emb", config.EMBED_MODEL + "::" + texts[i]).write_text(
                json.dumps(vecs[j].tolist()))
    return np.vstack(out)
```

```python
"""Embedding index + numpy cosine retrieval over the corpus KB."""
import numpy as np
import pandas as pd
from . import config, data_prep, llm_client

def _normalize(m: np.ndarray) -> np.ndarray:
    n = np.linalg.norm(m, axis=1, keepdims=True)
    n[n == 0] = 1.0
    return m / n

def build_index() -> None:
    config.ensure_dirs()
    vp = config.KB_DIR / "kb_vectors.npy"
    mp = config.KB_DIR / "kb_meta.parquet"
    if vp.exists() and mp.exists():
        return
    corpus, _ = data_prep.load_pools()
    vecs = llm_client.embed(corpus["customer_open"].tolist())
    np.save(vp, _normalize(vecs))
    corpus[["customer_open", "spotify_reply"]].reset_index(drop=True).to_parquet(mp)

def _load_index():
    vecs = np.load(config.KB_DIR / "kb_vectors.npy")
    meta = pd.read_parquet(config.KB_DIR / "kb_meta.parquet").to_dict("records")
    return vecs, meta

def retrieve(message: str, k: int = 4) -> list[dict]:
    vecs, meta = _load_index()
    q = _normalize(llm_client.embed([message]))[0]
    scores = vecs @ q
    idx = np.argsort(-scores)[:k]
    return [{**meta[i], "score": float(scores[i])} for i in idx]
```

```python
                          "hate the", "ui", "design", "terrible"],
}
# Order matters: more specific intents first (cancel before generic premium/billing).
_ORDER = ["cancellation_refund", "account_login", "billing_subscription",
          "playback_bug", "content_catalog", "feature_complaint"]

def weak_label(text: str) -> str:
    t = text.lower()
    for intent in _ORDER:
        for kw in KEYWORDS[intent]:
            if re.search(r"\b" + re.escape(kw), t):
                return intent
    return OTHER
```

```python
"""Trivial / simple (TF-IDF) / LLM classifiers."""
import json
from collections import Counter
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import Pipeline
from . import llm_client, data_prep, weak_labels, config
from .taxonomy import INTENT_NAMES, OTHER, describe

class TrivialClassifier:
    def __init__(self, majority: str):
        self.majority = majority
    def predict(self, msgs):
        return [self.majority] * len(msgs)

def fit_trivial(train_labels):
    majority = Counter(train_labels).most_common(1)[0][0]
    return TrivialClassifier(majority)

class SimpleClassifier:
    def __init__(self, pipe: Pipeline):
        self.pipe = pipe
    def predict(self, msgs):
        return list(self.pipe.predict(msgs))
    @classmethod
    def from_weak_corpus(cls):
        corpus, _ = data_prep.load_pools()
        X = corpus["customer_open"].tolist()
        y = [weak_labels.weak_label(t) for t in X]
        pipe = Pipeline([
            ("tfidf", TfidfVectorizer(ngram_range=(1, 2), min_df=2, max_features=20000)),
            ("clf", LogisticRegression(max_iter=1000, class_weight="balanced",
                                       random_state=config.SEED)),
        ])
        pipe.fit(X, y)
        return cls(pipe)

_CLS_PROMPT = """You are an intent classifier for Spotify customer-support tweets.
Classify the message into exactly one intent from this taxonomy:
{taxonomy}

Respond ONLY with JSON: {{"intent": "<one intent name>", "confidence": <0.0-1.0>}}.

Message: {message}"""

def llm_classify(message: str):
    raw = llm_client.generate(
        _CLS_PROMPT.format(taxonomy=describe(), message=message), json_mode=True)
    try:
        obj = json.loads(raw)
        intent = str(obj.get("intent", OTHER))
        conf = float(obj.get("confidence", 0.0))
    except (json.JSONDecodeError, ValueError, TypeError):
        intent, conf = OTHER, 0.0
    if intent not in INTENT_NAMES:
        intent = OTHER
    return intent, max(0.0, min(1.0, conf))
```

```python
from .retrieve import retrieve
from .taxonomy import INTENT_NAMES

CANNED = {
    "playback_bug": "Sorry for the trouble! Please DM us your device, OS, and app version so we can dig in. ^S",
    "account_login": "We can help you get back in. Please DM us the email on the account (never your password). ^S",
    "billing_subscription": "Thanks for flagging — please DM us the email on the account and we'll review the charge. ^S",

def trivial_reply(intent: str) -> str:

def nearest_reply(message: str) -> str:
    hits = retrieve(message, k=1)
    return hits[0]["spotify_reply"] if hits else CANNED["other"]

_GEN_PROMPT = """You are a Spotify customer-support agent on Twitter. Write ONE short,
empathetic reply (<280 chars) to the customer message below. Ground your reply in how

account-specific facts. End with the agent signature " ^S".

Predicted intent: {intent}

Historical examples (customer -> Spotify reply):
{examples}

Customer message: {message}

Reply:"""

    prompt = _GEN_PROMPT.format(intent=intent, examples=ex_block, message=message)
```

```python
"""Auditable rule-based escalation policy. Every path returns a reason string."""
import re

ESCALATE_INTENTS = {"account_login", "billing_subscription", "cancellation_refund"}
CONF_THRESHOLD = 0.55
MAX_UNRESOLVED_TURNS = 6  # 3+ back-and-forth rounds
_RISK = re.compile(r"\b(sue|lawsuit|legal|fraud|scam|gdpr|police|hack(ed|ing)?|"
                   r"stolen|unauthoriz|dispute|chargeback)\b", re.I)

def decide(intent: str, confidence: float, turns: list[dict], message: str):
    if _RISK.search(message or ""):
        return True, "Escalate: message contains legal/security/risk language."
    if intent in ESCALATE_INTENTS:
        return True, f"Escalate: '{intent}' is a sensitive intent (account/billing/cancellation)."
    if confidence < CONF_THRESHOLD:
        return True, f"Escalate: classifier confidence {confidence:.2f} below {CONF_THRESHOLD}."
    if turns and len(turns) >= MAX_UNRESOLVED_TURNS:
        return True, "Escalate: long unresolved back-and-forth thread (3+ rounds)."
    return False, f"Auto-handle: '{intent}' with confidence {confidence:.2f}, no risk signals."
```

```python
"""Build the golden eval set: stratified sample + LLM pre-labels for author review."""
from pathlib import Path
import pandas as pd
from support_agent import config, data_prep
from support_agent.classify import llm_classify
from support_agent.escalate import decide
from support_agent.taxonomy import INTENT_NAMES

def stratified_sample(eval_df, size, min_per_intent):
    df = eval_df.copy().reset_index(drop=True)
    preds = [llm_classify(m) for m in df["customer_open"]]
    df["pre_intent"] = [p[0] for p in preds]
    df["pre_confidence"] = [p[1] for p in preds]
    picked = []
    # guarantee the minimum per observed intent, then fill the rest at random
    for intent in df["pre_intent"].unique():
        grp = df[df["pre_intent"] == intent]
        take = min(len(grp), min_per_intent)
        picked.append(grp.sample(take, random_state=config.SEED))
    base = pd.concat(picked)
    remaining = df.drop(base.index)
    need = size - len(base)
    if need > 0 and len(remaining) > 0:
        base = pd.concat([base, remaining.sample(min(need, len(remaining)),
                                                 random_state=config.SEED)])
    return base.sample(min(size, len(base)), random_state=config.SEED).reset_index(drop=True)

def build() -> Path:
    config.ensure_dirs()
    out = config.GOLDEN_DIR / "golden_eval.csv"
    if out.exists():
        return out
    _, eval_df = data_prep.load_pools()
    g = stratified_sample(eval_df, config.GOLDEN_SIZE, config.GOLDEN_MIN_PER_INTENT)
    dec = [decide(r.pre_intent, r.pre_confidence, [], r.customer_open)
           for r in g.itertuples()]
    g["pre_escalate"] = [d[0] for d in dec]
    g["pre_reason"] = [d[1] for d in dec]
    g = g.rename(columns={"customer_open": "message"})
    g["gold_intent"] = g["pre_intent"]       # author edits these in place
    g["gold_escalate"] = g["pre_escalate"]
    g["gold_reason"] = g["pre_reason"]
    g["in_spotcheck"] = False
    sc = g.sample(min(40, len(g)), random_state=config.SEED).index
    g.loc[sc, "in_spotcheck"] = True
    cols = ["root_id","message","pre_intent","pre_confidence","pre_escalate",
            "pre_reason","gold_intent","gold_escalate","gold_reason","in_spotcheck"]
    g[cols].to_csv(out, index=False)
    return out

if __name__ == "__main__":
    print("Wrote", build())
```

```python
"""Gemini LLM-as-judge: score a drafted reply 1-5 on a fixed rubric."""
import json
from support_agent import llm_client

_RUBRIC = """You are evaluating a Spotify support reply. Score 1-5 (integers) on:
- grounded: consistent with how Spotify historically replied (reference shown)
- factual: makes no invented/unsupported claims
- tone: empathetic, on-brand, concise
- actionable: gives the customer a concrete next step
- overall: holistic quality
Reference (a real historical Spotify reply): {reference}
Customer message: {message}
Drafted reply: {reply}
Respond ONLY with JSON keys grounded, factual, tone, actionable, overall."""

_KEYS = ("grounded", "factual", "tone", "actionable", "overall")

def judge_reply(message: str, reply: str, reference: str) -> dict:
    raw = llm_client.generate(
        _RUBRIC.format(reference=reference, message=message, reply=reply),
        json_mode=True, temperature=0.0)
    try:
        obj = json.loads(raw)
    except json.JSONDecodeError:
        obj = {}
    out = {}
    for k in _KEYS:
        try:
            v = int(round(float(obj.get(k, 3))))
        except (ValueError, TypeError):
            v = 3
        out[k] = max(1, min(5, v))
    return out
```

```python
"""Automated eval harness: classification, escalation, reply-quality metrics."""
import json
import pandas as pd
from sklearn.metrics import accuracy_score, f1_score, precision_score, recall_score
from support_agent import config, classify, draft_reply, retrieve
from support_agent.classify import llm_classify
from eval.llm_judge import judge_reply

def classification_metrics(y_true, y_pred) -> dict:
    return {"accuracy": float(accuracy_score(y_true, y_pred)),
            "macro_f1": float(f1_score(y_true, y_pred, average="macro", zero_division=0))}

def escalation_metrics(y_true, y_pred) -> dict:
    return {"precision": float(precision_score(y_true, y_pred, zero_division=0)),
            "recall": float(recall_score(y_true, y_pred, zero_division=0))}

def main() -> dict:
    g = pd.read_csv(config.GOLDEN_DIR / "golden_eval.csv")
    msgs = g["message"].tolist()
    y_true = g["gold_intent"].tolist()

    trivial = classify.fit_trivial(y_true)
    simple = classify.SimpleClassifier.from_weak_corpus()
    llm_pred = [llm_classify(m)[0] for m in msgs]

    results = {"classification": {
        "trivial": classification_metrics(y_true, trivial.predict(msgs)),
        "simple_tfidf": classification_metrics(y_true, simple.predict(msgs)),
        "llm_fewshot": classification_metrics(y_true, llm_pred)}}

    # reply quality on a capped subset to keep the demo fast
    sub = g.head(min(60, len(g)))
    ref = sub["message"].map(lambda m: retrieve.retrieve(m, k=1)[0]["spotify_reply"])
    def avg_overall(replies):
        scores = [judge_reply(m, r, rf)["overall"]
                  for m, r, rf in zip(sub["message"], replies, ref)]
        return sum(scores) / len(scores)
    results["reply_overall"] = {
        "trivial": avg_overall([draft_reply.trivial_reply(i) for i in sub["gold_intent"]]),
        "nearest": avg_overall([draft_reply.nearest_reply(m) for m in sub["message"]]),
        "grounded": avg_overall([draft_reply.grounded_reply(m, i)
                                 for m, i in zip(sub["message"], sub["gold_intent"])])}

    # escalation vs gold
    esc_pred, esc_true = [], g["gold_escalate"].astype(bool).tolist()
    from support_agent.escalate import decide
    for m, (intent, conf) in zip(msgs, [llm_classify(m) for m in msgs]):
        esc_pred.append(decide(intent, conf, [], m)[0])
    results["escalation"] = escalation_metrics(esc_true, esc_pred)

    (config.INTERIM_DIR / "eval_results.json").write_text(json.dumps(results, indent=2))
    _print_table(results)
    return results

def _print_table(r: dict):
    print("\n=== Classification (accuracy / macro-F1) ===")
    for k, v in r["classification"].items():
        print(f"  {k:14s} acc={v['accuracy']:.3f}  macroF1={v['macro_f1']:.3f}")
    print("\n=== Reply quality (judge overall, 1-5) ===")
    for k, v in r["reply_overall"].items():
        print(f"  {k:10s} {v:.2f}")
    e = r["escalation"]
    print(f"\n=== Escalation ===\n  precision={e['precision']:.3f}  recall={e['recall']:.3f}")

if __name__ == "__main__":
    main()
```

```python
"""Agreement between the LLM judge and a human rater on reply overall scores."""
import pandas as pd
from scipy.stats import spearmanr
from sklearn.metrics import cohen_kappa_score
from support_agent import config

def _bin(s: int) -> int:
    return 0 if s <= 2 else (1 if s == 3 else 2)

def agreement(human, judge) -> dict:
    hb = [_bin(x) for x in human]
    jb = [_bin(x) for x in judge]
    rho = spearmanr(human, judge).correlation if len(human) > 1 else float("nan")
    return {"cohen_kappa": float(cohen_kappa_score(hb, jb)),
            "spearman": float(rho), "n": len(human)}

def main():
    hp = config.GOLDEN_DIR / "human_scores.csv"
    if not hp.exists():
        print("No human_scores.csv yet — score the in_spotcheck rows first.")
        return
    df = pd.read_csv(hp)   # cols: message, human_overall, judge_overall
    m = agreement(df["human_overall"].tolist(), df["judge_overall"].tolist())
    print(f"Judge vs human (n={m['n']}): Cohen's kappa={m['cohen_kappa']:.3f}, "
          f"Spearman={m['spearman']:.3f}")

if __name__ == "__main__":
    main()
```
